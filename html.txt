<script>
    let chart;
    const ctx = document.getElementById('graphCanvas').getContext('2d');

    // UI Elements
    const funcInput = document.getElementById('function-input');
    const boundAInput = document.getElementById('bound-a');
    const boundBInput = document.getElementById('bound-b');
    const resultDisplay = document.getElementById('result-display');
    const mathNotation = document.getElementById('math-notation');
    const errorMsg = document.getElementById('error-msg');

    // Simpson's Rule for Numerical Integration
    function integrate(f, a, b, n = 1000) {
        if (n % 2 !== 0) n++; // Simpson's rule requires an even number of intervals
        const h = (b - a) / n;
        let sum = f(a) + f(b);

        for (let i = 1; i < n; i++) {
            const x = a + i * h;
            sum += i % 2 === 0 ? 2 * f(x) : 4 * f(x);
        }
        return (h / 3) * sum;
    }

    function updateCalculator() {
        errorMsg.style.display = 'none';
        const exprString = funcInput.value;
        const a = parseFloat(boundAInput.value);
        const b = parseFloat(boundBInput.value);

        if (isNaN(a) || isNaN(b)) return;

        try {
            // Compile the function using math.js
            const compiledExpr = math.compile(exprString);
            const f = (x) => compiledExpr.evaluate({ x: x });

            // Test evaluation to catch math errors early
            f(a);

            // Calculate Integral
            const result = integrate(f, a, b);
            resultDisplay.textContent = Number(result.toFixed(5)).toString();
            mathNotation.innerHTML = `∫<sub>${a}</sub><sup>${b}</sup> (${exprString}) dx`;

            // Setup Graph Data boundaries (pad slightly wider than bounds a and b)
            const margin = Math.max(Math.abs(b - a) * 0.3, 1);
            const minX = Math.min(a, b) - margin;
            const maxX = Math.max(a, b) + margin;
            const steps = 200;
            const stepSize = (maxX - minX) / steps;

            const lineData = [];
            const fillData = [];

            for (let i = 0; i <= steps; i++) {
                const x = minX + i * stepSize;
                let y;
                try {
                    y = f(x);
                } catch {
                    y = null;
                }

                if (typeof y === 'number' && !isNaN(y) && isFinite(y)) {
                    lineData.push({ x: x, y: y });
                    
                    // Add to filled area if within integration bounds [a, b]
                    const currentLower = Math.min(a, b);
                    const currentUpper = Math.max(a, b);
                    if (x >= currentLower && x <= currentUpper) {
                        fillData.push({ x: x, y: y });
                    }
                }
            }

            // Update Chart.js Instance
            if (chart) chart.destroy();

            chart = new Chart(ctx, {
                type: 'line',
                data: {
                    datasets: [
                        {
                            label: 'f(x)',
                            data: lineData,
                            borderColor: '#3498db',
                            borderWidth: 2,
                            pointRadius: 0,
                            fill: false,
                            order: 1
                        },
                        {
                            label: 'Integrated Area',
                            data: fillData,
                            backgroundColor: 'rgba(52, 152, 219, 0.3)',
                            borderColor: 'transparent',
                            pointRadius: 0,
                            fill: 'origin', // fills down/up to the x-axis (y=0)
                            order: 2
                        }
                    ]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    scales: {
                        x: {
                            type: 'linear',
                            position: 'bottom',
                            grid: { color: '#e2e8f0' }
                        },
                        y: {
                            type: 'linear',
                            grid: { color: '#e2e8f0' }
                        }
                    },
                    plugins: {
                        legend: { display: false }
                    }
                }
            });

        } catch (error) {
            errorMsg.style.display = 'block';
            resultDisplay.textContent = "Error";
        }
    }

    // Attach Event Listeners for real-time calculation updates
    funcInput.addEventListener('input', updateCalculator);
    boundAInput.addEventListener('input', updateCalculator);
    boundBInput.addEventListener('input', updateCalculator);

    // Initial Run on Page Load
    updateCalculator();
</script>
